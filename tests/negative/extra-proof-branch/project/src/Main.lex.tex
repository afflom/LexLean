\title{Natural number addition}

\begin{theorem}{add-zero}
\noaxioms
For every natural number \(n\), \(n + 0 = n\) and \(n * 1 = n\).
\begin{proof}
\begin{constructor}
\begin{branch}{1}
Close the goal by reflexivity.
\end{branch}
\begin{branch}{2}
Close the goal by reflexivity.
\end{branch}
\begin{branch}{3}
Close the goal by reflexivity.
\end{branch}
\end{constructor}
\end{proof}
\end{theorem}
\end{lexlean}
